1. git version - проверка версии git
2. git config --global user.name "ваш_ник" и git config --global user.email "ваша_почта@yandex.ru" - указываем имя пользователя и эл. почту (--global, значит что применится ко всем git-репозиториям на данном компьютере) для подписи коммитов
3. cd Documents/ - перейти в папку документы
4. mkdir Lab2_ISRPO_FIO - создать папку лаб2
5. git init -b main - создаёт новый локальный Git-репозиторий, -b main указывает имя основной ветки.
6. git remote add origin ссылка_на_репозиторий - привязать удалённый репозиторий к локальному
7. git remote -v - проверить подключение удалённого репозитория
8. dotnet new console -o Program --use-program-main - создать консольное приложение с помощью .NET CLI
9. .gitignore - файл для исключения служебных файлов и папок из репозитория
10. git status - проверить состояние репозитория
11. git add . - добавить все файлы (кроме игнорируемых) в индекс
12. git commit -m "Initial commit: create console app" - коммит с описанием изменений
13. git log - команда для показа истории коммитов в текущей ветке
14. git push -u origin main - отправить изменения в удалённый репозиторий
15. git add Program/Program.cs - добавить изменения в индекс
16. git log --oneline - для быстрого просмотра, когда нужны только основные данные
17. git show ваш_хеш - 
18. git checkout -b new-feature - создание новой ветки с переключением (устаревший подход)
19. git branch hotfix - создание ветки без переключения
20. git branch - просмотр всех веток
21. git switch -c - создание ветки и переключение (новая команда)
22. git switch - переключение на другую ветку (новая команда)
23. git switch -  - переключение на предыдущую ветку
24. git branch -d hotfix - удалить ветку
25. git pull - обновляет main
26. git fetch --prune - актуальное состояние сервера (загружает информацию об удаленных ветках, --prune — удаляет ссылки на несуществующие удаленные ветки)
